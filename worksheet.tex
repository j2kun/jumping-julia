% Define command to draw a puzzle
\newcommand{\drawpuzzle}[2]{
\begin{tikzpicture}[scale=1.2]
  \foreach \row in {0,1,2,3} {
    \foreach \col in {0,1,2,3} {
      \pgfmathtruncatemacro{\idx}{\row*4+\col}
      \pgfmathsetmacro{\val}{#2[\idx]}

      % Determine fill color
      \ifnum\row=0
        \ifnum\col=0
          \def\fillcol{startgreen}
        \else
          \def\fillcol{white}
        \fi
      \else
        \ifnum\row=3
          \ifnum\col=3
            \def\fillcol{goalyellow}
          \else
            \def\fillcol{white}
          \fi
        \else
          \def\fillcol{white}
        \fi
      \fi

      % Draw square
      \draw[line width=2pt, fill=\fillcol] (\col*1.5, -\row*1.5) rectangle +(1.3, 1.3);

      % Add number or star
      \ifnum\row=3
        \ifnum\col=3
          \node at (\col*1.5+0.65, -\row*1.5+0.65) {\Huge $\bigstar$};
        \else
          \node at (\col*1.5+0.65, -\row*1.5+0.65) {\Huge\bfseries \pgfmathprintnumber[int trunc]{\val}};
        \fi
      \else
        \node at (\col*1.5+0.65, -\row*1.5+0.65) {\Huge\bfseries \pgfmathprintnumber[int trunc]{\val}};
      \fi

      % Add START label
      \ifnum\row=0
        \ifnum\col=0
          \node[font=\tiny\bfseries] at (\col*1.5+0.65, -\row*1.5+1.15) {START};
        \fi
      \fi
    }
  }
\end{tikzpicture}
}
